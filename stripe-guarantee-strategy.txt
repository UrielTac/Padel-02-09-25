### Fase 1: Endpoint de Cargo [2 horas]

```typescript
// 1. Crear nuevo endpoint: /api/stripe/charge-no-show/route.ts
import { stripe } from '@/lib/stripe';
import { createRouteHandlerClient } from '@supabase/auth-helpers-nextjs';

export async function POST(req: Request) {
  try {
    const { bookingId, reason, amount } = await req.json();
    const supabase = createRouteHandlerClient({ cookies });

    // 1.1 Obtener datos necesarios
    const [booking, stripeCustomer] = await Promise.all([
      getBookingDetails(supabase, bookingId),
      getStripeCustomer(supabase, bookingId)
    ]);

    // 1.2 Crear PaymentIntent
    const paymentIntent = await stripe.paymentIntents.create({
      amount: Math.round(amount * 100),
      currency: 'eur',
      payment_method: stripeCustomer.stripe_payment_method_id,
      off_session: true,
      confirm: true,
      metadata: {
        booking_id: bookingId,
        charge_type: 'no_show',
        reason
      }
    }, {
      stripeAccount: stripeCustomer.stripe_account_id
    });

    // 1.3 Registrar en base de datos
    await createNoShowPaymentRecord(supabase, {
      bookingId,
      paymentIntentId: paymentIntent.id,
      amount,
      reason
    });

    return NextResponse.json({ success: true });
  } catch (error) {
    return handleStripeError(error);
  }
}
```

### Fase 2: Servicio de Pagos [2 horas]

```typescript
// 2. Crear/Actualizar: src/services/stripePaymentService.ts
export class StripePaymentService {
  private stripe: Stripe;
  private supabase: SupabaseClient;

  constructor(stripe: Stripe, supabase: SupabaseClient) {
    this.stripe = stripe;
    this.supabase = supabase;
  }

  async chargeNoShow({
    bookingId,
    reason,
    amount
  }: NoShowChargeParams): Promise<PaymentResult> {
    // 2.1 Validaciones
    await this.validateNoShowCharge(bookingId);

    // 2.2 Procesar cargo
    const result = await this.processNoShowCharge({
      bookingId,
      amount,
      reason
    });

    // 2.3 Actualizar estado
    await this.updateBookingStatus(bookingId, {
      status: 'cancelled',
      reason,
      chargeApplied: true
    });

    return result;
  }

  private async validateNoShowCharge(bookingId: string) {
    const booking = await this.getBookingWithGuarantee(bookingId);
    if (!booking) throw new Error('Reserva no encontrada');
    if (booking.payment_type !== 'guarantee') {
      throw new Error('La reserva no tiene garantía');
    }
    // ... más validaciones
  }
}
```

### Fase 4: Webhooks y Notificaciones [2 horas]

```typescript
// 4.1 Actualizar: /api/stripe/webhooks/route.ts
export async function POST(req: Request) {
  const body = await req.text();
  const signature = headers().get('stripe-signature')!;

  try {
    const event = stripe.webhooks.constructEvent(
      body,
      signature,
      process.env.STRIPE_WEBHOOK_SECRET!
    );

    switch (event.type) {
      case 'payment_intent.succeeded':
        await handleSuccessfulNoShowCharge(event.data.object);
        break;
      case 'payment_intent.payment_failed':
        await handleFailedNoShowCharge(event.data.object);
        break;
    }

    return NextResponse.json({ received: true });
  } catch (error) {
    return handleWebhookError(error);
  }
}
```
